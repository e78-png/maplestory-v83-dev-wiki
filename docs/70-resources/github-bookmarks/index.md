let cat = "";
const qEl = document.getElementById('bmQ');
const langEl = document.getElementById('lang');
const verEl = document.getElementById('ver');
const ageEl = document.getElementById('age');
const main = document.getElementById('bmMain');
const nav = document.getElementById('bmNav');
const cnt = document.getElementById('bmCnt');

function esc(s){return (s||'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}

function buildNav(){
  nav.innerHTML = '<button class="bm-btn '+(cat===''?'on':'')+'" data-c="">全部</button>' +
    DATA.map(c=>'<button class="bm-btn '+(cat===c.key?'on':'')+'" data-c="'+c.key+'">'+esc(c.title)+' ('+c.items.length+')</button>').join('');
  nav.querySelectorAll('button').forEach(b=>b.onclick=()=>{cat=b.dataset.c;buildNav();render();window.scrollTo({top:0});});
}

function render(){
  const q = qEl.value.trim().toLowerCase();
  const lg = langEl.value;
  const vv = verEl.value;
  const ag = ageEl.value;
  let shown = 0;
  let html = '';
  for(const c of DATA){
    if(cat && c.key !== cat) continue;
    let items = c.items.filter(it=>{
      if(lg && it.lang !== lg) return false;
      if(vv && it.ver !== vv) return false;
      if(ag && it.age !== ag) return false;
      if(!q) return true;
      return (it.full_name+' '+it.desc+' '+it.note+' '+(it.lang||'')+' '+(it.ver||'')).toLowerCase().includes(q);
    });
    if(!items.length) continue;
    shown += items.length;
    html += '<section class="bm-sec"><h3>'+esc(c.title)+' <span class="n">'+items.length+' 個</span></h2>'
         + '<p class="bm-hint">'+esc(c.hint)+'</p><div class="grid">'
         + items.map(it=>{
             let t = '';
             if (it.stars) t += '<span class="bm-tag st">★ '+it.stars+'</span>';
             if (it.ver)  t += '<span class="bm-tag ver">'+esc(it.ver)+'</span>';
             if (it.lang) t += '<span class="tag">'+esc(it.lang)+'</span>';
             if (it.age) {
               const cls = it.age==='活躍'?'live':(it.age==='維護中'?'slow':(it.age==='已封存'?'arch':'dead'));
               t += '<span class="bm-tag '+cls+'">'+esc(it.age)+'</span>';
             }
             return '<div class="bm-card"><div class="bm-top">'
               + '<a class="repo" href="'+esc(it.url)+'" target="_blank" rel="noopener">'+esc(it.full_name)+'</a>'
               + '<span class="bm-tags">'+t+'</span></div>'
               + (it.desc?'<div class="bm-desc">'+esc(it.desc)+'</div>':'')
               + '<div class="bm-cn">'+esc(it.note)+'</div></div>';
           }).join('')
         + '</div></section>';
  }
  if(!shown) html = '<div class="empty">找不到符合的專案</div>';
  main.innerHTML = html;
  cnt.textContent = shown + ' / ' + DATA.reduce((a,c)=>a+c.items.length,0) + ' 個專案';
}
qEl.addEventListener('input', render);
langEl.addEventListener('change', render);
verEl.addEventListener('change', render);
ageEl.addEventListener('change', render);
buildNav(); render();
})();
